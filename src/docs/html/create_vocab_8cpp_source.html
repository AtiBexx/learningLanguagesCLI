203// Kezeli az "exit" parancsot a fájlnév és mappanevek bekérésekor
204// Handles the "exit" command when prompted for file and folder names
205FileCreationAction handleFileCreationInput(const std::string& input) {
206 std::string cleanInput = toLowerCase(trim(input));
207 if (cleanInput == "exit") {
208 return FileCreationAction::EXIT;
209 }
210 // Megkeressük az első szóközt (pl. "del mappa" -> szóköz a 3. indexen)
211 // Find the first space (e.g. "del folder" -> space at index 3)
212 size_t spacePos = cleanInput.find(' ');
213
214 if (spacePos != std::string::npos) {
215
216 // A szóköz előtti rész a parancs (pl. "del")
217 // The part before the space is the command
218 std::string cmd = toLowerCase(trim(input.substr(0, spacePos)));
219
220 // A szóköz utáni rész az útvonal (pl. "Uj mappa") - Ezt NEM toLowerCase-ezzük!
221 // The part after the space is the path (e.g. "New folder") - We do NOT toLowerCase this!
222 std::string rawPath = trim(input.substr(spacePos + 1));
223
224 // ./data előtag hozzáadása
225 // Add ./data prefix
226 std::string path = "./data/";
227 path.append(rawPath);
228
229 // --- TÖRLÉS (del / rm) ---
230 // DEL / RM
231 if (cmd == "del" || cmd == "rm") {
232 bool isDir = isDirectory(path); // A C++11-es stat alapú függvényed
233 deleteFileFolders(path, isDir); // A cliFileReading-ben lévő függvényed
234 return FileCreationAction::COMMAND_EXECUTED;
235 }
236 // --- ÁTNEVEZÉS ---
237 // REN / RN
238 if (cmd == "rn") {
239 bool isDir = isDirectory(path);
240 reNameFileFolders(path, isDir);
241 return FileCreationAction::COMMAND_EXECUTED;
242 }
243 // --- MÁSOLÁS / MOZGATÁS (hasonlóan...) ---
244 // CP / MV
245 if (cmd == "cp") {
246 copyFileFolders(path, isDirectory(path));
247 return FileCreationAction::COMMAND_EXECUTED;
248 }
249 // MOZGATÁS
250 // MV --MOVE--
251 if (cmd == "mv") {
252 movingFileFolders(path, isDirectory(path));
253 return FileCreationAction::COMMAND_EXECUTED;
254 }
255 }
256 return FileCreationAction::CONTINUE;
257}

260int handleInput(const std::string& input, std::ofstream& file, bool allowEmpty) {
261
262 // Kilépés kezelése
263 // Exit handling
264 if (input == "exit" || input == "Exit" || input == "EXIT" ) {
265 file.flush();
266 wordsPairTotal(true);
267 return 2; // Jelzés a break-re
268 }
269
270 // Csak mentés kezelése
271 // Only save handling
272 if (input == "save" || input == "Save" || input == "SAVE") {
273 file.flush();
274 wordsPairTotal(false);
275 return 1; // Jelzés a continue-ra
276 }
277
278 // VISSZA FUNKCIÓ (ha elrontotta az első szót)
279 // UNDO FUNCTION (if you messed up the first word)
280 if (input == "back" || input == "Back" || input == "BACK") {
281 return 1;
282 }
283
284 // Ha üres ÉS nem engedélyezett az üres, akkor újrakezdjük
285 // If empty AND empty is not allowed, then restart
286 if (input.empty() && !allowEmpty) {
287 return 1;
288 }
289
290 return 0; // Minden OK, mehetünk a következő bekérésre
291}
