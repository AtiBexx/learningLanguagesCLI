94// function to trim spaces (trim)
95string trim(const std::string& str) {
96 //megkeressük az első karaktert ami nem szóköz
97 size_t first = str.find_first_not_of(" \t\n\r");
98 //std::string::npos az jelenti nincs megtalálva
99 if (first == std::string::npos) {
100 return("");
101 }else {
102 size_t last = str.find_last_not_of(" \t\n\r");
103 return str.substr(first, (last - first + 1));
104 }
105}

123// Load words from file
124vector<WordPair> loadWords(const std::string &filename) {
125 vector<WordPair> words;
126 ifstream file(filename);
127 string line;
128
129 if (!file.is_open()) {
130 const W_wordPair &wWordPair = wordPairErrorTranslations [static_cast<int>(programUiLanguage)];
131 logError("loadWords", wWordPair.ErrorOpenedFile);
132
133 std:: cerr << wWordPair.ErrorOpenedFile <<" "<<filename << std::endl;
134 return words;
135 }
136 while (getline(file, line)) {
137 if (line.empty()) continue;
138
139 WordPair currentWord;
140
141 //Megkeresük a kiejtést ha van [ ] között
142 //Find the pronunciation if it is between [ ]
143 size_t openBracket = line.find('[');
144 size_t closeBracket = line.find(']' , openBracket);
145
146 if (openBracket != string::npos && closeBracket != string::npos) {
147 currentWord.pronunciation = line.substr(openBracket + 1, closeBracket - openBracket - 1);
148 currentWord.pronunciation = trim(currentWord.pronunciation);
149 line = line.substr(0, openBracket);
150 }
151
152 //Megkeresük az elválasztó vonalat
153 //Find the dividing line
154 size_t separatorPos = line.find("->");
155
156 // 2 karaktert ugrunk mert"->" ez 2 karakter és ez után olvasunk
157 // we skip 2 characters because "->" is 2 characters and we read after that
158 int skipChars = 2;
159
160 // Ha nincs meg megkeressük ezt '>' ez "->" helyett
161 // If it doesn't exist, we'll look for this '>' instead of "->"
162 if (separatorPos == string::npos) {
163 separatorPos = line.find('>');
164 skipChars = 1; //itt csak egyet ugrunk || just one jumping here
165 }
166
167 if (separatorPos != string::npos) {
168 currentWord.targetLangMeaning = line.substr(0, separatorPos);
169
170 size_t hungarianStart = separatorPos + skipChars;
171 if (hungarianStart < line.length()) {
172 currentWord.motherLangMeaning = line.substr(hungarianStart);
173 }
174 //Trim
175 currentWord.targetLangMeaning = trim(currentWord.targetLangMeaning);
176 currentWord.motherLangMeaning = trim(currentWord.motherLangMeaning);
177
178 // Ha az angol szó és a magyar szó se üres akkor rakjuk be a vektorba
179 // If neither the English word nor the Hungarian word is empty, insert it into the vector
180 if (!currentWord.targetLangMeaning.empty() && !currentWord.motherLangMeaning.empty()) {
181 words.push_back(currentWord);
182 }
183 }
184 }
185 file.close();
186 return words;
187}
