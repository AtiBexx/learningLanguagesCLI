178//To remove punctuation, ignore '!' '?' etc.
179std::string removePunctuation(const std::string& s)
180{
181 std::string result;
182 for (char c : s) {
183 // Ha nem írásjel, akkor megtartjuk
184 // If it's not a punctuation mark, we keep it
185 if (c != '!' && c != '?' && c != '.' && c != ',' && c != ';') {
186 result += c;
187 }
188 }
189 return result;
190}
191// Az Ékezetes karatkterek figyelmen kívül hagyása
192// Ignore Accented Characters
193std::string removeAccents(const std::string &s) {
194 // UTF-8 ékezetes karakterek és párjaik
195 static const std::map<std::string, std::string> accentMap = {
196 {"á", "a"}, {"é", "e"}, {"í", "i"}, {"ó", "o"}, {"ö", "o"}, {"ő", "o"},
197 {"ú", "u"}, {"ü", "u"}, {"ű", "u"},
198 {"Á", "a"}, {"É", "e"}, {"Í", "i"}, {"Ó", "o"}, {"Ö", "o"}, {"Ő", "o"},
199 {"Ú", "u"}, {"Ü", "u"}, {"Ű", "u"}
200 };
201
202 std::string result;
203 for (size_t i = 0; i < s.length(); ) {
204 bool found = false;
205 // Megnézzük, hogy az aktuális pozíciótól kezdődik-e egy 2-bájtos ékezetes karakter
206 // Check if the current position starts with a 2-byte accented character
207 /*
208 * Miért ilyen bonyolult a típus?
209 std::pair: A map minden eleme egy pár (kulcs és érték).
210 const std::string (az első): A map kulcsa sosem változhat meg utó lag.
211 std::string (a második): Ez a sima betű , amit hozzáadunk.
212 &: Referencia, hogy ne másoljuk le az elemeket minden körben.
213 */
214 // for (auto const& pair : accentMap)
215 for (const std::pair<const std::string, std::string>& pair : accentMap) {
216 const std::string& accent = pair.first;
217 const std::string& plain = pair.second;
218
219 if (s.compare(i, accent.length(), accent) == 0) {
220 result += plain;
221 i += accent.length();
222 found = true;
223 break;
224 }
225 }
226 if (!found) {
227 result += s[i]; //Hozzáadjuk az a betűt az eredményhez
228 i++; // Továbblépünk a következő karakterre
229 }
230 }
231 return result;
232}
